// Injected by backend/tools/docsgql (apicompat export) into backend/core/natsauth
// through `go test -overlay`; it is never part of the tree on disk. It asks the broker
// grant builder what ONE device may publish to and subscribe to, and writes it, as the
// MQTT topics a device uses, to $DC_APICOMPAT_OUT. The grant is the topic layout
// itself: it is what the broker enforces, built from the same declarations the ingest
// subscription and the subject parsers use.

package natsauth_test

import (
	"encoding/json"
	"os"
	"sort"
	"strings"
	"testing"

	"github.com/devicechain-io/dc-microservice/messaging"
	"github.com/devicechain-io/dc-microservice/natsauth"
)

func TestAPICompatExportTopics(t *testing.T) {
	out := os.Getenv("DC_APICOMPAT_OUT")
	if out == "" {
		t.Skip("DC_APICOMPAT_OUT is not set; this test is the exporter apicompat injects")
	}
	p, err := natsauth.DevicePermissions("INSTANCE", "TENANT", "DEVICE")
	if err != nil {
		t.Fatal(err)
	}
	topics := func(subjects []string) []string {
		var ts []string
		for _, s := range subjects {
			// "$MQTT.sub.>" is the broker's own delivery plumbing, not a topic a device
			// addresses.
			if strings.HasPrefix(s, "$") {
				continue
			}
			s = messaging.SubjectToMqttTopic(s)
			s = strings.NewReplacer("INSTANCE", "{instance}", "TENANT", "{tenant}", "DEVICE", "{device}").Replace(s)
			ts = append(ts, s)
		}
		sort.Strings(ts)
		return ts
	}
	b, err := json.MarshalIndent(map[string]any{
		"publish":   topics(p.Pub.Allow),
		"subscribe": topics(p.Sub.Allow),
	}, "", "  ")
	if err != nil {
		t.Fatal(err)
	}
	if err := os.WriteFile(out, append(b, '\n'), 0o644); err != nil {
		t.Fatal(err)
	}
}
